    cudaDriverEntryPointQueryResult q{};

struct VRange {
    CUdeviceptr base = 0;
    size_t reserved = 0, mapped = 0;
    std::vector<CUmemGenericAllocationHandle> h;
    void* ptr() const { return (void*) base; }
    static size_t up(size_t b) { return (b + g_vmm.gran - 1) / g_vmm.gran * g_vmm.gran; }
    void init(size_t bytes) {
        reserved = up(std::max<size_t>(bytes, 1));
        if (g_vmm.reserve(&base, reserved, 0, 0, 0) != CUDA_SUCCESS) {
            std::fprintf(stderr, "cuMemAddressReserve of %zu bytes failed\n", reserved);
            std::exit(1);
        }
    }
    size_t need(size_t bytes) const { return std::min(up(bytes), reserved); }   // mapped bytes for `bytes`
    bool map_to(size_t bytes) {   // false: out of memory (what was mapped stays)
        const size_t want = need(bytes);
        CUmemAllocationProp prop{};
        prop.type = CU_MEM_ALLOCATION_TYPE_PINNED;
        prop.location.type = CU_MEM_LOCATION_TYPE_DEVICE;
        prop.location.id = g_vmm.dev;
        CUmemAccessDesc acc{};
        acc.location = prop.location;
        acc.flags = CU_MEM_ACCESS_FLAGS_PROT_READWRITE;
        while (mapped < want) {
            CUmemGenericAllocationHandle hd = 0;
            if (g_vmm.create(&hd, g_vmm.gran, &prop, 0) != CUDA_SUCCESS) return false;
            if (g_vmm.map(base + mapped, g_vmm.gran, 0, hd, 0) != CUDA_SUCCESS ||
                g_vmm.access(base + mapped, g_vmm.gran, &acc, 1) != CUDA_SUCCESS) {
                std::fprintf(stderr, "cuMemMap failed\n");
                std::exit(1);
            }
            h.push_back(hd);
            mapped += g_vmm.gran;
            g_vram += g_vmm.gran;
        }
        return true;
    }
    void unmap_to(size_t bytes) {
        const size_t want = need(bytes);
        while (mapped > want) {
            mapped -= g_vmm.gran;
            g_vmm.unmap(base + mapped, g_vmm.gran);
            g_vmm.release(h.back());
            h.pop_back();
            g_vram -= g_vmm.gran;
        }
    }

    std::vector<int> order;

        for (int e = 0; e < NE; ++e) order.push_back(l * NE + e);

            ++evicted;
        }
    };

    evict(owner_mx);

                const int candidate = order[next++], l = candidate / NE;
                if (ex_nv(l) == nv && desired[candidate] && slot_of[candidate] < 0) { key = candidate; break; }

            std::memcpy(packed.data() + p.w13_bytes + p.s13_bytes, p.w2, p.w2_bytes);

            owners[s] = key;
            bytes += eb;

struct PoolShrinkPlan {
    int drop_nv = 0;
    int drop_mx = 0;
    bool fits = true;
};
PoolShrinkPlan pool_shrink_plan(double bytes) {
    PoolShrinkPlan p;
    if (bytes <= 0) return p;
    double deficit = bytes + 4.0 * g_vmm.gran;
    for (bool nv : {false, true}) {
        const int ns = nv ? E.nslot_nv : E.nslot_mx;
        if (!ns) continue;
        const size_t eb = nv ? ex_bytes(0) : MX_BYTES;
        const int floor = FETCH_MAX;
        const int drop = std::min(ns - floor, (int) std::ceil(deficit / eb));
        if (nv) p.drop_nv = drop;
        else p.drop_mx = drop;
        deficit -= (double) drop * eb;
    }
    p.fits = deficit <= 0;
    return p;
}
// release outer MXFP8 slots first to preserve the loop-layer NVFP4 pool
void pools_release(double bytes, const char* what) {
    const PoolShrinkPlan p = pool_shrink_plan(bytes);
    if (!p.fits) {
        std::fprintf(stderr, "%s would violate expert pool floors (NVFP4 %d, MXFP8 %d)\n", what,
                     FETCH_MAX, FETCH_MAX);
        std::exit(1);
    }
    if (p.drop_mx) pool_resize(false, E.nslot_mx - p.drop_mx);
    if (p.drop_nv) pool_resize(true, E.nslot_nv - p.drop_nv);
}
// the pools take what is free beyond the margin: first the configured MXFP8 share, then NVFP4
void pools_fill(bool refill = true) {
    if (!E.offload || !E.grow) return;
    const int old_nv = E.nslot_nv, old_mx = E.nslot_mx;
    const size_t keep = vram_keep() + 4 * g_vmm.gran;
    if (E.nslot_mx < E.mx_target && E.vpool_mx.reserved) {
        const size_t fr = vram_free();
        if (fr > keep) {
            const int cap = std::min(E.mx_target, (int) (E.vpool_mx.reserved / MX_BYTES));
            const int add = std::min(cap - E.nslot_mx, (int) ((fr - keep) / MX_BYTES));
            if (add > 0) pool_resize(false, E.nslot_mx + add);
        }
    }
    const size_t fr = vram_free();
    if (fr > keep) {
        const int cap = (int) (E.vpool_nv.reserved / ex_bytes(0));
        const int add = std::min(cap - E.nslot_nv, (int) ((fr - keep) / ex_bytes(0)));
        if (add > 0) pool_resize(true, E.nslot_nv + add);
    }
    if (refill && (E.nslot_nv > old_nv || E.nslot_mx > old_mx)) {
        std::fprintf(stderr, "pools: %d NVFP4 + %d MXFP8 slots (%.2f GiB)\n", E.nslot_nv, E.nslot_mx,
                     (E.nslot_nv * (double) ex_bytes(0) + E.nslot_mx * (double) MX_BYTES) / 1073741824.0);
        refill_lazy_pool();
    }
}
// the full slots hold `need` cells (grown in --kv-grow-step steps, from the expert pools when VRAM is short)
void kv_fit(int need) {
    if (!E.grow || need <= E.kv_cells) return;
    const int nc = std::min(E.ctx, (need + E.grow_step - 1) / E.grow_step * E.grow_step);
    CK(cudaDeviceSynchronize());
    const size_t add = kv_mapped_for(nc) - kv_mapped_for(E.kv_cells), keep = vram_keep(), fr = vram_free();
    if (fr < add + keep && E.offload) pools_release((double) (add + keep - fr), "the K/V");
    for (size_t i = 0; i < g_kvr.size(); ++i)
        if (!g_kvr[i].map_to((size_t) nc * g_kvr_bpc[i])) { std::fprintf(stderr, "--kv-grow: out of VRAM at %d cells\n", nc); std::exit(1); }
    E.kv_cells = nc;
    std::fprintf(stderr, "kv-grow: %d cells (%.2f GiB K/V), expert slots %d + %d\n", nc, kv_mapped_for(nc) / 1073741824.0,
                 E.nslot_nv, E.nslot_mx);
}
// a request that needs only `need` cells: the rest of the mapped K/V back to the pools
void kv_trim(int need) {
    if (!E.grow) return;
    const int nc = std::min(E.ctx, std::max(E.grow_step, (need + E.grow_step - 1) / E.grow_step * E.grow_step));
    if (E.kv_cells <= nc + E.grow_step) return;
    CK(cudaDeviceSynchronize());
    for (size_t i = 0; i < g_kvr.size(); ++i) g_kvr[i].unmap_to((size_t) nc * g_kvr_bpc[i]);
    E.kv_cells = nc;
    pools_fill();
    std::fprintf(stderr, "kv-grow: back to %d cells, expert slots %d + %d\n", nc, E.nslot_nv, E.nslot_mx);
}

// A chunk is read routed only if its pools can hold all its ids at once, so that no group's plan evicts an expert an
// earlier group of the chunk fetched (its tiles would be skipped): T x TOPK, plus the promotions in flight with the
// CPU experts, within the smaller pool.  CBA: counts ids, not distinct experts (a device count would cost a sync).
bool pf_routed_ok(int T) {

// the prompt buffers for chunks of T tokens (T <= --chunk): mapped, from the pools if VRAM is short
void pf_map(int T) {
    if (g_pfr.empty() || T <= g_pf_mapped_T) return;
    CK(cudaDeviceSynchronize());
    size_t add = 0;
    std::vector<size_t> want(g_pfr.size());
    for (size_t i = 0; i < g_pfr.size(); ++i) {
        const VRange& r = g_pfr[i];
        want[i] = g_pfr_whole[i] == 2 ? (pf_routed_ok(T) ? 0 : r.reserved)
                : g_pfr_whole[i] ? r.reserved : r.need(std::min(r.reserved, (size_t) ((double) r.reserved * (T + 64) / PF.TP)));
        if (want[i] > r.mapped) add += want[i] - r.mapped;
    }
    const size_t keep = vram_keep(), fr = vram_free();
    if (fr < add + keep) pools_release((double) (add + keep - fr), "the prompt buffers");
    for (size_t i = 0; i < g_pfr.size(); ++i)
        if (!g_pfr[i].map_to(want[i])) { std::fprintf(stderr, "prompt buffers: out of VRAM\n"); std::exit(1); }
    g_pf_mapped_T = T;
}

// the largest chunk (<= T, halving down to 256) whose prompt buffers and K/V growth leave each pool its FETCH_MAX
// slots: long contexts on a small card read the prompt in smaller chunks instead of stopping (g_pf_full: not even 256)
int pf_fit(int T, int pos0) {
    if (g_pfr.empty() || !E.offload) return T;
    for (;; T /= 2) {
        size_t want = 0, have = 0;
        for (size_t i = 0; i < g_pfr.size(); ++i) {
            const VRange& r = g_pfr[i];
            want += g_pfr_whole[i] == 2 ? (pf_routed_ok(T) ? 0 : r.reserved)
                  : g_pfr_whole[i] ? r.reserved : r.need(std::min(r.reserved, (size_t) ((double) r.reserved * (T + 64) / PF.TP)));
            have += r.mapped;
        }
        const int nc = std::min(E.ctx, (pos0 + T + E.grow_step - 1) / E.grow_step * E.grow_step);
        const double kv_add = E.grow && nc > E.kv_cells ? (double) (kv_mapped_for(nc) - kv_mapped_for(E.kv_cells)) : 0.0;
        const double deficit = kv_add + (double) want - (double) have + (double) vram_keep() - (double) vram_free() + 8.0 * g_vmm.gran;
        const bool fits = pools_keep_min(deficit);
        if (fits || T <= 256) { g_pf_full = !fits; return T; }
    }
}

// Offloaded prefill: the expert cache is frozen during a prefill (a chunk touches nearly every expert, an LRU would
// only thrash).  A grouped product runs once over the resident experts (the decode tables, null = not resident:
// tiles skipped), then over the others in batches streamed into two staging halves on PF.cs (copy of batch b + 1
// overlapping the product of batch b).  Each row still sees exactly its expert's weights: bit-identical.
// part B (w2) is streamed after the SwiGLU, separately from part A (w13): each byte crosses PCIe once per layer call.
// The regular lazy prefill streams every nonresident expert. DSpark verification streams only its routed misses.
template <class F> void expert_parts(int l, bool partB, F launch) {
    const Experts& X = E.L[l].ex;
    launch(partB ? X.tB : X.tA);
    if (!E.offload || g_pf_routed || (PF.rows_argmax && E.cpu)) return;
    const auto& nonres = PF.verify_route_misses ? PF.verify_nonres : PF.nonres[l];
    const size_t eb = ex_bytes(l), part = partB ? eb - ex_partA(l) : ex_partA(l), off = partB ? ex_partA(l) : 0;
    const int per = (int) (PF.stg_half / part);
    for (int b0 = 0, bi = 0; b0 < (int) nonres.size(); b0 += per, ++bi) {
        const int h = PF.stg_turn++ & 1, n = std::min(per, (int) nonres.size() - b0);
        CK(cudaStreamWaitEvent(PF.cs, PF.ev_free[h], 0));
        CK(cudaEventSynchronize(PF.ev_ready[h]));   // the previous table copy out of ptab[h] is done
        const uint8_t** tab = PF.ptab[h];
        std::fill(tab, tab + NE, nullptr);
        for (int j = 0; j < n; ++j) {
            const int e = nonres[b0 + j];
            uint8_t* d = PF.stg[h] + (size_t) j * part;
            if (E.lazy_experts) {
                const ink2cpu::ExpertParts& p = E.expert_parts[(size_t) l * NE + e];
                uint8_t* src = PF.hstg[h] + (size_t) j * part;
                const uint8_t* w = partB ? p.w2 : p.w13;
                const uint8_t* s = partB ? p.s2 : p.s13;
                const size_t nw = partB ? p.w2_bytes : p.w13_bytes;
                const size_t ns = partB ? p.s2_bytes : p.s13_bytes;
                if (nw + ns != part) { std::fprintf(stderr, "layer %d expert %d: prefill source size mismatch\n", l, e); std::exit(1); }
                std::memcpy(src, w, nw);
                std::memcpy(src + nw, s, ns);
                CK(cudaMemcpyAsync(d, src, part, cudaMemcpyHostToDevice, PF.cs));
            } else {
                CK(cudaMemcpyAsync(d, E.xh[(size_t) l * NE + e] + off, part, cudaMemcpyHostToDevice, PF.cs));
            }
            tab[e] = d;
        }
        // the batch table rides on the copy stream too (pinned slot per half, reused once ev_free[h] has fired)
        CK(cudaMemcpyAsync(PF.dtab[h], PF.ptab[h], NE * sizeof(void*), cudaMemcpyHostToDevice, PF.cs));
        CK(cudaEventRecord(PF.ev_ready[h], PF.cs));
        CK(cudaStreamWaitEvent(cudaStreamPerThread, PF.ev_ready[h], 0));
        launch(PF.dtab[h]);
        CK(cudaEventRecord(PF.ev_free[h], cudaStreamPerThread));
        PF.fbytes += (size_t) n * part;
    }
}

__device__ __forceinline__ float head_norm_rope(float v, const float* __restrict__ w, int pos) {
    __shared__ float red[4], sm[HD];
    const int d = threadIdx.x;
    float ss = warp_sum(v * v);
    __syncthreads();
    if ((d & 31) == 0) red[d >> 5] = ss;
    __syncthreads();

bool pools_keep_min(double bytes) {

// can the K/V grow to `need` cells (kv_fit's step) without a pool falling under its minimum?
bool kv_room(int need) {
    if (!E.grow || need <= E.kv_cells) return true;

// The conversation cache of --serve (--conversation-cache-mib N, off by default): agentic use (one agent over many
// turns, an agent and its sub-agents taking turns) does not read again a prompt it has already read.
//
// The slots hold one conversation (the live one, `hist` of the serve loop).  Its state after n tokens is, per layer
// visit: the full-attention K / V rows [0, n) (8 visits), the sliding-window rows of positions [n - WINDOW, n) in
// their ring (40 visits), the four causal-conv rings (khist / vhist / ahist / mhist, HR rows each, 48 visits); with
// --spec the drafter's K / V (its global layer: rows [0, n), its two local layers: their ring's window); and the
// logits after token n - 1.  Every --kv format is bytes per row here (codes and scales alike, kv_cell_bytes).  Not
// per conversation, not saved: the expert pools and the CPU experts' placement, the drafter's Markov context (the
// serve loop always reads at least the prompt's last token with --spec, which recomputes it), the sampler.
//
// A checkpoint (Ck) is the "tail" of a state at n: everything above except the full-attention rows, ~190 MiB at f16,
// in page-locked host memory.  The full rows [0, n) are free to keep: a later position never writes them.  So:
//   - at each prompt's end (before the first sampled token) a checkpoint of the live conversation: a client that
//     sends the history back re-rendered (its reasoning dropped, its tool call re-serialized) diverges from the
//     generated tokens after the prompt, and the next request restores that point instead of reading from 0;
//   - a request that does not extend the live conversation's last prompt parks it (its full rows, its checkpoints
//     and one at its history end) and restores the parked conversation, live checkpoint or branch point that covers
//     the longest prefix of the request;
//   - a request sharing a long prefix with a known conversation without a checkpoint there (sub-agents sharing a
//     system prompt and tools) stops its prefill at the branch point once and takes a checkpoint there.
// LRU over whole parked conversations under the budget.  Off: none of this runs (the serve loop is the base one).

    g_sess.min_tokens = min_tokens;

        }

// positions [lo, hi) of an array <-> host bytes h (position order)
void sess_rows(const SessArr& a, int lo, int hi, uint8_t* h, bool d2h) {
    for (int q = lo; q < hi;) {
        const int r = q % a.cap, m = std::min(hi - q, a.cap - r);
        uint8_t* dv = a.d + (size_t) r * a.b;
        uint8_t* hv = h + (size_t) (q - lo) * a.b;
        CK(cudaMemcpyAsync(d2h ? (void*) hv : (void*) dv, d2h ? (const void*) dv : (const void*) hv, (size_t) m * a.b,
                           d2h ? cudaMemcpyDeviceToHost : cudaMemcpyHostToDevice, cudaStreamPerThread));
        q += m;
    }
}

    size_t t = 0;
    auto tail = [&](const PinnedP& p) {
        if (p && std::find(seen.begin(), seen.end(), p.get()) == seen.end()) { seen.push_back(p.get()); t += p->n; }
    };
    for (auto& c : g_sess.live) tail(c.tail);
    for (auto& v : g_sess.parked) { t += v.full ? v.full->n : 0; for (auto& c : v.cks) tail(c.tail); }
    for (auto& p : g_sess.spare) tail(p);
    return t;
}
// room for `bytes` more: spare buffers freed first, then the least recently used parked conversations (not one held)
bool sess_room(size_t bytes) {
    while (sess_used() + bytes > g_sess.budget) {
        if (!g_sess.spare.empty()) { g_sess.spare.pop_back(); continue; }
        auto lru = g_sess.parked.end();
        for (auto it = g_sess.parked.begin(); it != g_sess.parked.end(); ++it)
            if (!it->hold && (lru == g_sess.parked.end() || it->used < lru->used)) lru = it;
        if (lru == g_sess.parked.end()) return false;
        g_sess.note += " evicted " + std::to_string(lru->ids.size());
        g_sess.parked.erase(lru);
    }
    return true;
}
// CBA: page-locked buffers are kept for reuse (cudaHostAlloc pins ~0.45 ms per MiB on a PCIe 3 host: 80 ms per
// checkpoint), a spare taken for anything up to half its size; checkpoints in PIN_UNIT steps, a parked conversation's
// rows in PIN_UNIT << k (a conversation that grew a turn finds its last buffer again).  Up to half a row buffer is
// slack; a sub-allocator in one pinned arena would waste less
constexpr size_t PIN_UNIT = (size_t) 32 << 20;
PinnedP sess_alloc(size_t bytes, bool geo = false) {
    bytes = (bytes + PIN_UNIT - 1) / PIN_UNIT * PIN_UNIT;
    if (geo) { size_t g = PIN_UNIT; while (g < bytes) g *= 2; bytes = g; }
    auto& sp = g_sess.spare;
    size_t bi = sp.size();
    for (size_t i = 0; i < sp.size(); ++i)
        if (sp[i]->n >= bytes && sp[i]->n <= 2 * bytes && (bi == sp.size() || sp[i]->n < sp[bi]->n)) bi = i;
    if (bi < sp.size()) { PinnedP p = sp[bi]; sp.erase(sp.begin() + bi); return p; }
    if (!sess_room(bytes)) return nullptr;
    PinnedP p = std::make_shared<Pinned>(bytes);
    return p->p ? p : nullptr;
}
void sess_drop(PinnedP&& p) {   // a buffer nothing uses any more goes to the spares
    if (p && p.use_count() == 1) g_sess.spare.push_back(std::move(p));
    p.reset();
}
// a checkpoint of the live state at n (its logits: lg)
bool sess_take(int n, const vec& lg, int kind) {
    PinnedP t = sess_alloc(g_sess.tail_bytes);
    if (!t) { g_sess.note += " no-room"; return false; }
    sess_tail(n, t->p, true);
    g_sess.live.push_back({n, t, lg, kind});
    return true;
}

// The live conversation (hist, last_logits) to the parked set, when worth it: its full rows, its checkpoints and its
// history end.  The live checkpoints stay valid for the slots (a later restore may use them).
void sess_park(const std::vector<int>& hist, const vec& last_logits) {
    if ((int) hist.size() < g_sess.min_tokens) return;
    Conv v;
    v.ids = hist;
    v.prompt_n = g_sess.live_prompt;
    v.cks = g_sess.live;
    const bool end = std::none_of(v.cks.begin(), v.cks.end(), [&](const Ck& c) { return c.n == (int) hist.size(); });
    const size_t need = hist.size() * g_sess.row_bytes + (end ? g_sess.tail_bytes : 0);
    if (!sess_room(need)) { g_sess.note += " park-no-room"; return; }
    // in the parked set (held) while its buffers are taken, so that the budget counts them
    v.used = ++g_sess.clock;
    v.hold = true;
    const uint64_t id = v.used;
    g_sess.parked.push_back(std::move(v));
    auto me = [&]() -> Conv& { for (auto& c : g_sess.parked) if (c.used == id) return c; return g_sess.parked.back(); };
    PinnedP full = sess_alloc(hist.size() * g_sess.row_bytes, true);
    if (!full) {
        for (size_t j = 0; j < g_sess.parked.size(); ++j) if (g_sess.parked[j].used == id) g_sess.parked.erase(g_sess.parked.begin() + j);
        g_sess.note += " park-no-room";
        return;
    }
    me().full = full;
    sess_full((int) hist.size(), full->p, true);
    if (end) {
        PinnedP t = sess_alloc(g_sess.tail_bytes);
        if (t) { sess_tail((int) hist.size(), t->p, true); me().cks.push_back({(int) hist.size(), t, last_logits, CK_END}); }
    }
    me().hold = false;
    g_sess.note += " parked " + std::to_string(hist.size());
}

// Before the request's kv_trim (a parked conversation's rows are still mapped): picks the longest prefix of pr with a
// state, parks the live conversation if pr does not continue it, restores that state.  `start` = the base loop's
// choice (hist.size() when pr extends hist, else 0); returns the position to read from.  hist / last_logits follow.
size_t sess_begin(const std::vector<int>& pr, std::vector<int>& hist, vec& last_logits, size_t start) {
    g_sess.split = -1;
    g_sess.note.clear();
    const double t0 = now_ms();
    // (a picture's image tokens are ids made from its pixels' hash, ink2_vision.cuh vis_id: every comparison below
    // tells two pictures of the same size apart)
    const size_t l_live = sess_lcp(hist, pr);
    // the best state: the live history end (start), a live checkpoint, a parked conversation's checkpoint
    size_t best = start, share = l_live;
    int src = start > 0 ? 0 : -1, ck = -1, pv = -1;   // src: 0 live end, 1 live checkpoint, 2 parked, -1 none
    for (size_t i = 0; i < g_sess.live.size(); ++i)
        if (g_sess.live[i].n <= (int) l_live && g_sess.live[i].n > (int) best) { best = g_sess.live[i].n; src = 1; ck = (int) i; }
    for (size_t j = 0; j < g_sess.parked.size(); ++j) {
        const size_t l = sess_lcp(g_sess.parked[j].ids, pr);
        share = std::max(share, l);
        for (size_t i = 0; i < g_sess.parked[j].cks.size(); ++i)
            if (g_sess.parked[j].cks[i].n <= (int) l && g_sess.parked[j].cks[i].n > (int) best) { best = g_sess.parked[j].cks[i].n; src = 2; ck = (int) i; pv = (int) j; }
    }
    // the live conversation is parked unless the request extends its last prompt (the same conversation, re-rendered
    // or continued)
    const bool same = (!hist.empty() && l_live == hist.size()) || (g_sess.live_prompt > 0 && (int) l_live >= g_sess.live_prompt);
    // a branch point between conversations only (a re-render diverging deep in the generated tail is not one)
    if (!same && share >= best + SPLIT_MIN && share < pr.size()) g_sess.split = (int) share;
    if (pv >= 0) g_sess.parked[pv].hold = true;
    if (!same && !hist.empty()) sess_park(hist, last_logits);   // (may evict parked conversations, not the held one)
    for (size_t j = 0; j < g_sess.parked.size(); ++j) if (g_sess.parked[j].hold) { pv = (int) j; g_sess.parked[j].hold = false; }
    const char* how = "live";
    if (src == 1) {   // the slots' own rows [0, n): the tail only
        const Ck c = g_sess.live[ck];
        sess_tail(c.n, c.tail->p, false);
        last_logits = c.logits;
        how = "rewind";
    } else if (src == 2) {
        Conv& v = g_sess.parked[pv];
        const Ck c = v.cks[ck];
        kv_fit(c.n + 1);   // --kv-grow: a conversation longer than the K/V mapped now
        if ((int) sess_lcp(hist, v.ids) < c.n || same) {   // (same: the slots' rows past the live prompt may differ)
            // the parked rows array after array: n rows of each, from an image of ids.size() rows each
            CK(cudaDeviceSynchronize());
            const uint8_t* h = v.full->p;
            for (auto& a : g_sess.full) { sess_rows(a, 0, c.n, const_cast<uint8_t*>(h), false); h += v.ids.size() * a.b; }
            CK(cudaDeviceSynchronize());
        }
        sess_tail(c.n, c.tail->p, false);
        last_logits = c.logits;
        how = "parked";
        v.used = ++g_sess.clock;
        // the request continues that conversation (extends its last prompt): it leaves the parked set, its checkpoints
        // up to n become the live ones; else (a sibling from a shared branch point) it stays parked, its branch
        // checkpoints shared
        std::vector<Ck> keep;
        for (auto& x : v.cks) if (x.n <= c.n && (x.n == c.n || x.kind == CK_BRANCH)) keep.push_back(x);
        g_sess.live = keep;
        if (v.prompt_n > 0 && (int) best >= v.prompt_n) {   // its buffers to the spares (unless shared with the live)
            Conv dead = std::move(v);
            g_sess.parked.erase(g_sess.parked.begin() + pv);
            sess_drop(std::move(dead.full));
            for (auto& x : dead.cks) sess_drop(std::move(x.tail));
        }
    } else if (src < 0) {
        how = "none";
    }
    for (size_t i = g_sess.live.size(); i-- > 0;)   // checkpoints past the read start are no longer the slots' state
        if (g_sess.live[i].n > (int) best) { PinnedP t = g_sess.live[i].tail; g_sess.live.erase(g_sess.live.begin() + i); sess_drop(std::move(t)); }
    g_sess.live_prompt = 0;   // (sess_prompt_end sets it)
    hist.assign(pr.begin(), pr.begin() + best);
    std::fprintf(stderr, "[ink2] conversation cache: prompt %zu, read from %zu (%s)%s%s, %zu parked, %.0f of %zu MiB, %.0f ms\n",
                 pr.size(), best, how, g_sess.split > 0 ? (", branch point " + std::to_string(g_sess.split)).c_str() : "",
                 g_sess.note.c_str(), g_sess.parked.size(), sess_used() / 1048576.0, g_sess.budget >> 20, now_ms() - t0);
    return best;
}

// the prefill chunk at p0 of T tokens: ends at the branch point when it crosses it
int sess_chunk(int p0, int T) { return g_sess.on && g_sess.split > p0 && g_sess.split < p0 + T ? g_sess.split - p0 : T; }
// after the chunk ending at p: the branch checkpoint.  CBA: the live conversation's checkpoints are never evicted
// (only parked conversations are); one branch point per request at most, they leave with it when it is parked
void sess_after_chunk(int p) {
    if (!g_sess.on || p != g_sess.split) return;
    vec lg(VOCAB);
    CK(cudaMemcpy(lg.data(), E.logits, (size_t) VOCAB * 4, cudaMemcpyDeviceToHost));
    const double t0 = now_ms();
    if (sess_take(p, lg, CK_BRANCH)) std::fprintf(stderr, "[ink2] conversation cache: branch checkpoint at %d, %.0f ms\n", p, now_ms() - t0);
}

// tools/inkling2/ink2_cpu.hpp - Inkling-2's cold experts on the CPU.
//
// The decode graph hands the experts of a layer visit that are not in VRAM to the CPU through one doorbell per node
// (InkBell, in that node's memory, registered mapped): the GPU writes the visit's input x, the jobs and a sequence
// number into each; one pool of pinned threads per NUMA node computes the jobs whose expert lives on that node
// (gate_up, softcapped SwiGLU, down: the same math as the GPU kernels; fp32 activations, int8 opt-in) and writes each
// expert's output row and its ack into its own doorbell; the graph waits for both acks, then the MoE tail weights all
// eight outputs as usual.
//
// Experts live in host memory placed per node (ink2cpu::node_alloc), each expert whole on one node, and so do the
// doorbells and the pools' buffers: a node's threads only touch local memory.  On a two-socket box the inter-socket
// link is the slow path; only the GPU's traffic (doorbells, expert copies) crosses it. INK2_NUMA_POOL=0 uses one pool
// over all cores and leaves expert memory unplaced.
#pragma once
#include <cstddef>
#include <cstdint>

namespace ink2cpu {

constexpr int BELL_H = 2304, BELL_T = 8, BELL_K = 8 * BELL_T;
struct InkBell {   // one per node in its memory, registered mapped; GPU writes ring/job/x, the node's pool y/ack
    volatile unsigned ring;            // sequence number of the last visit with CPU jobs
    unsigned pad0[15];
    volatile unsigned ack[2];          // [node]: the last sequence number done (written in that node's bell)
    volatile float us_job;             // the pool's time per job, us (moving average; the GPU's PCIe share reads it)
    unsigned pad1[13];
    int layer, n, t;                   // the visit's layer, number of jobs and input rows
    int k[BELL_K], e[BELL_K], count[BELL_K];
    int row[BELL_K][BELL_T], slot[BELL_K][BELL_T]; // selected inputs and output slots for each expert job
    unsigned pad2[14];
    alignas(64) float x[BELL_T][BELL_H]; // the visit's MoE inputs
    alignas(64) float y[BELL_K][BELL_H]; // job j's expert output (factor applied, not the router weight), in the bell
                                       // of its expert's node
};

struct ExpertParts {
    const uint8_t* w13 = nullptr;
    const uint8_t* s13 = nullptr;
    const uint8_t* w2 = nullptr;
    const uint8_t* s2 = nullptr;
    size_t w13_bytes = 0;
    size_t s13_bytes = 0;
    size_t w2_bytes = 0;
    size_t s2_bytes = 0;
};

struct Setup {
    int nl = 0, ne = 0, mx_first = 0, h = 0, ff = 0;
    const uint8_t* const* ex = nullptr;    // [nl * ne] host address of expert (l, e)'s slot
    const ExpertParts* ex_parts = nullptr; // split tensors in the read-only checkpoint mapping
    const uint8_t* home = nullptr;          // [nl * ne] its node
    const float* f13 = nullptr;             // [nl * ne] NVFP4 per-expert factors (layers < mx_first)
    const float* f2 = nullptr;
    size_t nv_partA = 0, mx_partA = 0;      // offset of part B (w2 | s2) in a slot
    float gate_cap = 0, up_cap = 0;
    InkBell* bell[2] = {};                  // [node]
    int threads = 0;                        // per node
};

int nodes();                                // NUMA nodes with CPUs (1 when there is no NUMA information)
// bytes of anonymous memory on `node`, its pages touched by a thread pinned there (so they are placed there)
void* node_alloc(size_t bytes, int node);
void node_free(void* p, size_t bytes);
const char* isa();                          // "avx512" | "avx2" | nullptr (no usable kernel: AVX2 + FMA + F16C needed)

void start(const Setup& s);                 // starts the pools; they spin on the bell (and nap when it is idle)
void stop();

`INK2_CPU_OPT=29`, `INK2_OFF_MARGIN_MIB=64`, `INK2_MX_POOL_FRACTION=0.08`), defaults to 5 CPU
threads, uses `--spec 0`, i8r KV unless server flags select another supported cache format, and a
4096-token chunk. It picks a route profile beside the checkpoint:
`inkling2-routes-best512.freq` for a 16K prompt plus 512 output (`-c 16896`) and
`inkling2-routes-256k-decode.freq` otherwise. Stale `INK2_SPEC_DRAFT_N_MAX`, `INK2_SPEC_FORCE`,
`INK2_SPEC_DIAG`, and `INK2_ROUTE_PROFILE` values from an earlier shell are ignored so a DSpark
experiment cannot silently slow a normal request. Set `INK2_ALLOW_TUNING_OVERRIDE` to opt back into
environment overrides; use `bel-inkling2 --spec 1 --spec-draft-n-max N` for DSpark tests. Direct
mode supports one active text request and temperature, top-k, top-p, min-p, seed, `ignore_eos`, and
`cache_prompt`, which defaults to enabled. Keep `cache_prompt` enabled for production-style runs.
`--cache-type-k q8_0 --cache-type-v q4_0` selects direct K8/V4 KV; the direct engine always uses
tiled CUDA attention, including when `-fa on` is supplied. It also copies up to 512 MiB of
route-ranked nonresident experts from the safetensors mapping into a pinned host pool, capped by
available cgroup headroom and a 768 MiB safety reserve; colder experts remain file-backed. Set
both `INK2_ALLOW_TUNING_OVERRIDE=1` and `INK2_PIN_HOST_MIB=N` to change that budget. Grammar, logit bias,
repetition/frequency/presence penalties, mirostat, dry, and similar sampling options return an
unsupported error. No images, sleep/reload, external draft models, or parallel requests.

The embedded engine uses `ink2::embedded_server` for its line protocol, so a rebuild keeps
`bel-server` and `bel-cli` working. The checkpoint reader is shared with the reference executable.
